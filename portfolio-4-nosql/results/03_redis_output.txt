
# Key-value model (Redis 7): data that is hot, short-lived and accessed by a single key.

# Run: redis-cli < 03_redis_keyvalue.redis
127.0.0.1:6379> FLUSHDB
    OK

# 1) Shopping cart = HASH per customer (field = product_id, value = qty). Expires after 7 days of inactivity.
127.0.0.1:6379> HSET cart:4242 prod:101 2 prod:205 1
    2
127.0.0.1:6379> HINCRBY cart:4242 prod:101 1
    3
127.0.0.1:6379> EXPIRE cart:4242 604800
    1
127.0.0.1:6379> HGETALL cart:4242
    prod:101
    3
    prod:205
    1
127.0.0.1:6379> TTL cart:4242
    604800

# 2) Login session = STRING with 30-minute TTL
127.0.0.1:6379> SET session:abc123 "customer:4242" EX 1800
    OK
127.0.0.1:6379> GET session:abc123
    customer:4242
127.0.0.1:6379> TTL session:abc123
    1800

# 3) Flash sale: 3 units of product 9001. DECR is atomic, so concurrent buyers cannot oversell.
127.0.0.1:6379> SET flash:stock:9001 3
    OK
127.0.0.1:6379> DECR flash:stock:9001
    2
127.0.0.1:6379> DECR flash:stock:9001
    1
127.0.0.1:6379> DECR flash:stock:9001
    0
127.0.0.1:6379> DECR flash:stock:9001
    -1

# a negative result means sold out -> the app restores it with INCR and rejects the order
127.0.0.1:6379> INCR flash:stock:9001
    0

# 4) Real-time best-sellers = SORTED SET (score = units sold today)
127.0.0.1:6379> ZINCRBY bestsellers:today 5 "prod:101"
    5
127.0.0.1:6379> ZINCRBY bestsellers:today 12 "prod:205"
    12
127.0.0.1:6379> ZINCRBY bestsellers:today 7 "prod:330"
    7
127.0.0.1:6379> ZINCRBY bestsellers:today 3 "prod:101"
    8
127.0.0.1:6379> ZREVRANGE bestsellers:today 0 2 WITHSCORES
    prod:205
    12
    prod:101
    8
    prod:330
    7
127.0.0.1:6379> ZREVRANK bestsellers:today "prod:101"
    1
